#!/usr/local/bin/foma -f
# Rule-based Proto-Celtic → daughter transducers. xd xd xd 
#
# Daughters: Old_Irish, Middle_Welsh, Old_Welsh, Old_Breton, Middle_Breton,
#            Old_Cornish, Middle_Cornish, Gaulish, Celtiberian.
#
# Modelled on the structure of germanic.txt. Proto-Celtic segments use the
# {*x} brace notation throughout. Plain input characters are mapped to their
# starred counterparts by the pcelt* template definitions, and the starred
# forms are consumed by language-specific rule chains.
#
# Proto-Celtic consonantism: p, b, t, d, k, g, kʷ, gʷ, s, x, m, n, l, r, y,
#   w, ɸ.  No /f/ (labiodental) and no /ō/ (PIE *ō merged with *ā in PC);
#   both are daughter-language innovations or rule outputs.
#
# Phonological rules are simplified approximations. Complex conditioned changes
# (morphological lenition classes, OI epenthesis, Brittonic affection,
# Gaulish paradigm-specific apocope, etc.) produce known mismatches that are
# documented in the comparison report.

# 0a. Universal pre-rule: initial *u before *ɸ behaves as *w in Insular Celtic
#     (*uɸo → fo/guo). Applied before OIInitialW / BritInitialW.
define PCeltInitialUPhi [ {*u} -> {*w} || .#. _ {*ɸ} ];

# 0b. Universal: *ɸ → 0 in all daughters.
define PCeltPhiLoss [ {*ɸ} -> 0 ];

# 1. Labiovelars are Q-Celtic (merged with plain velars in Goidelic).
define OILabioVelar [
    {*kʷ} -> {*k},
    {*gʷ} -> {*g}
];

# 2. Initial *gd- cluster: *g drops (*gdonyos > duin).
define OIGdCluster [ {*g} {*d} -> {*d} || .#. _ ];

# 3. Initial *sf- cluster: *f drops (*sɸerā after phi-loss → *sera > ser).
define OISFCluster [ {*s} {*f} -> {*s} || .#. _ ];

# 4. Initial *w > *f (*wiros > fer, *windos > find).
define OIInitialW [ {*w} -> {*f} || .#. _ ];

# 5. *w after a consonant > 0 (*ardwos > ard).
define OIMedialWDrop [ {*w} -> 0 || PCeltStarConsonant _ ];

# 6. *w after a vowel > *u (*biwos > beo, *abū > auv).
define OIMedialWVocal [ {*w} -> {*u} || PCeltStarVowel _ ];

# 7. *y > 0 everywhere in Old Irish.
define OIYDrop [ {*y} -> 0 ];

# 8. Nasal-anterior mutations. Word-final *ants > *a (drops *nts). Medial *ant
#    before a vowel > *ād (short) or *ēd (long); both paths kept for ambiguity.
define OINasalAnt [
    {*a}{*n}{*t}{*s} -> {*a} || _ .#.,
    {*a}{*n}{*t} -> ({*ē} d | {*a} d) || _ PCeltStarVowel,
    {*a}{*n}{*k} -> ({*ē} g | {*a} g) || _ PCeltStarVowel
];

# 9. *mb > *m (*kambos > cam).
define OIMbSimpl [ {*m} {*b} -> {*m} ];

# 10. *u > *o before *k + following vowel (*drukos > droch).
define OIULowering [ {*u} -> {*o} || _ PCeltStarConsonant {*o} ];

# 11. *s > 0 between vowels (*esoxs > eo).
define OIMedialSLoss [ {*s} -> 0 || PCeltStarVowel _ PCeltStarVowel ];

# 12. *t > *θ between vowels (*latis > lath, *brātīr > brāθīr).
define OIMedialT [ {*t} -> {*θ} || PCeltStarVowel _ PCeltStarVowel ];

# 13. *d > *ð between vowels.
define OIMedialD [ {*d} -> {*ð} || PCeltStarVowel _ PCeltStarVowel ];

# 14. *k > *x after vowel/sonorant, before vowel or word-finally.
define OIFinalK [
    {*k} -> {*x} || PCeltStarLenCtx _ PCeltStarVowel,
    {*k} -> {*x} || PCeltStarLenCtx _ .#.
];

# 15. *b > *v between vowels (*abū > auv).
define OIBLenition [ {*b} -> {*v} || PCeltStarVowel _ PCeltStarVowel ];

# 16. Diphthong mergers in Old Irish.
define OIVowelDiphthongs [
    {*ei} -> {*ī},
    {*ai} -> {*e},
    {*au} -> {*o},
    {*ou} -> {*ū},
    {*ow} -> {*ū}
];

# 17. Nasal assimilation: *n > *ŋ before *g (*longā > loŋg).
#     Must run AFTER lenition (rules 11–15) and BEFORE *g spirantisation (18–20)
#     so *g is still present when *n assimilates. *ŋ is NOT in PCeltStarLenCtx,
#     so *ŋg clusters survive spirantisation intact.
define OINasalAssimN [ {*n} -> {*ŋ} || _ {*g} ];

# 18. *g > *ɣ between vowels, or after *r before a vowel (*lugus>luɣ,
#     *argantom>arɣad). *l+g (*bolgos>bolg) and *ŋ+g (*longā>loŋg) do NOT
#     spirantise — *l is excluded here, *ŋ is excluded by OINasalAssimN.
define OIMedialG [
    {*g} -> {*ɣ} || PCeltStarVowel _ PCeltStarVowel,
    {*g} -> {*ɣ} || {*r}          _ PCeltStarVowel
];

# 19. *g > *ɣ word-finally after any vowel.
define OIFinalG [ {*g} -> {*ɣ} || PCeltStarVowel _ .#. ];

# 20. *g > 0 word-finally after a long vowel (*rīgos→rī).
define OIGLossAfterLong [ {*g} -> 0 || PCeltStarLongVowel _ .#. ];

# 21. Final-syllable apocope (three sequential passes).
#     Pass 2 drops only *s (not *m) after a vowel: handles *trīs → trī etc.
#     *m is NOT dropped here — stem-final *m from OIMbSimpl must survive.
define OIApocope [
    [{*o}{*s} -> 0, {*o}{*m} -> 0, {*u}{*s} -> 0, {*x}{*s} -> 0 || _ .#.]
    .o.
    [{*s} -> 0 || PCeltStarVowel _ .#.]
    .o.
    [{*ā} -> 0, {*ī} -> 0, {*u} -> 0, {*i} -> 0 || _ .#.]
];

# 22. Strip asterisks; map long vowels to colon notation.
define OIRemoveStars [
    {*a} -> a,       {*ā} -> ā,
    {*e} -> e,       {*ē} -> ē,
    {*i} -> i,       {*ī} -> (ī | i),
    {*o} -> o,       {*ū} -> ū,
    {*u} -> u,
    {*b} -> b,       {*d} -> d,
    {*f} -> f,       {*g} -> g,
    {*h} -> h,       {*k} -> k,
    {*l} -> l,       {*m} -> m,
    {*n} -> n,       {*p} -> p,
    {*r} -> r,       {*s} -> s,
    {*t} -> t,       {*v} -> v,
    {*w} -> u,       {*x} -> x,
    {*y} -> 0,
    {*θ} -> θ,       {*ð} -> ð,
    {*ɣ} -> ɣ,       {*χ} -> χ,
    {*ŋ} -> ŋ
];

##############################################
## Brittonic transducer                      ##
##############################################

# 1. *kʷ > *p, *gʷ > *b (the defining P-Celtic innovation).
define BritLabioVelar [
    {*kʷ} -> {*p},
    {*gʷ} -> {*b}
];

# 2. Initial *gd- > *d (*gdonyos > din).
define BritGdCluster [ {*g} {*d} -> {*d} || .#. _ ];

# 2b. Initial *skʷ- > *χw- BEFORE BritLabioVelar converts *kʷ→*p,
#     so *skʷetlom → *χwetlom (MW chwedl) not *spetlom (OW).
define BritSkwCluster [ {*s} {*kʷ} -> {*χ} {*w} || .#. _ ];

# 3. Initial consonant-cluster simplifications before other initial-s rules.
#    *sr- > *fr- (shared with Gaulish; *srognā > froin/frogna).
#    *sw- > *χw- (MW chw-), *sf- > *f-, *sɸ- > *f- (*sɸerā > fer), *sm- > *m-, *sl- > *l-.
define BritInitialSClusters [
    {*s} {*r} -> {*f} {*r} || .#. _,
    {*s} {*w} -> {*χ} {*w} || .#. _,
    {*s} {*f} -> {*f}       || .#. _,
    {*s} {*ɸ} -> {*f}       || .#. _,
    {*s} {*m} -> {*m}       || .#. _,
    {*s} {*l} -> {*l}       || .#. _
];

# 4. Remaining initial *s > *h.
define BritInitialS [ {*s} -> {*h} || .#. _ ];

# 5. Initial *w > *gw (*wiros > gwr, *windos > gwinn).
#    gw appears as g+u in the IPA tokens (OW guo, MW guið etc.).
define BritInitialW [ {*w} -> {*gw} || .#. _ ];

# 6. *w after a consonant > 0 (*ardwos > arð).
define BritMedialWDrop [ {*w} -> 0 || PCeltStarConsonant _ ];

# 7. *y > *i word-initially before a vowel (MW iant/yantus, iaul/yālos);
#    *y > 0 elsewhere (medial *y in *olyos > ol).
define BritYDrop [
    {*y} -> {*i} || .#. _ PCeltStarVowel,
    {*y} -> 0
];

# 8. *ū > *i (short i; Brittonic u-raising: *dūnom > din, *rūnā > rin).
#    Produces short surface i, not long ī, matching all TSV daughter forms.
define BritUtoI [ {*ū} -> {*i} ];

# 9. *mb > *m (*kambos > camm/cam in Brittonic).
define BritMbSimpl [ {*m} {*b} -> {*m} ];

# 10. Diphthong mergers in Brittonic.
#     *au > *u (OW bu from *baus); *ow > *aw or *u (naw/*rud from *nowan/*rowdos).
define BritVowelDiphthongs [
    {*ei} -> {*oi},
    {*ai} -> {*oi},
    {*au} -> {*u},
    {*ou} -> {*u},
    {*ow} -> ({*aw} | {*u})
];

# 10b. I-affection (umlaut): *a → *i in syllable before *ī (e.g. *bleidanī > bloiðin).
#      Applied before apocope so the conditioning *ī is still present.
define BritIAffection [ {*a} {*n} {*ī} -> {*i} {*n} {*ī} ];

# 11. Final-syllable apocope (mirrors OIApocope; pass 2 drops *s only).
#     Pass 3 does NOT drop *o: *uɸo → guo (final *o is stem, not paradigm ending).
define BritApocope [
    # retain final *us to allow reconstruction (e.g. *bitus > bid)
    [{*o}{*s} -> 0, {*o}{*m} -> 0 || _ .#.]
    .o.
    [{*s} -> 0 || PCeltStarVowel _ .#.]
    .o.
    [{*t}{*u} -> {*t} || _ .#.]
    .o.
    [{*ā} -> 0, {*i} -> 0 || _ .#.]
    .o.
    # Drop *ī word-finally only in multi-syllabic forms (V C+ before *ī),
    # so *bloidinī > bloiðin but monosyllabic *trī (< *trīs) keeps its *ī.
    [{*ī} -> 0 || PCeltStarVowel PCeltStarConsonant+ _ .#.]
    .o.
    # *xs > *χ word-finally (MW χwēχ from *swexs; *natrixs > natriχ).
    [{*x}{*s} -> {*χ} || _ .#.]
    .o.
    # *ns > *s word-finally (*mīns > mīs, *karants partial).
    [{*n}{*s} -> {*s} || _ .#.]
    .o.
    # *ants > 0 word-finally (*karants > kar, *tarants, *karant...).
    [{*a}{*n}{*t}{*s} -> 0 || _ .#.]
    .o.
    # Drop final *m after *ā (e.g. *ɸlāmā > *lāmā > *lām > *lā > lau).
    [{*m} -> 0 || {*ā} _ .#.]
];

# 12. Lenition of obstruents in weak position. Each rule is a separate define
#     composed sequentially to avoid parallel-replacement + .#. interactions.
# Medial *xs → *χ before vowel (*ouxselos > uχel).
define BritMedialXS [ {*x}{*s} -> {*χ} || _ PCeltStarVowel ];
# *sn cluster simplification (*kʷresnom > pren; *s drops before nasal).
define BritSNSimpl [ {*s}{*n} -> {*n} ];
# *ī > 0 before word-final *r (r-stem syncope: *brātīr > brotr).
define BritIRSyncope [ {*ī} -> 0 || _ {*r} .#. ];
define BritLenInterD [ {*d} -> {*ð} || PCeltStarLenCtx _ PCeltStarVowel ];
define BritLenFinalD [ {*d} -> {*ð} || PCeltStarLenCtx _ .#. ];
define BritLenT     [ {*t} -> {*d} || PCeltStarVowel _ PCeltStarSonorant ];
define BritLenFinalK  [ {*k} -> {*χ} || PCeltStarSonorant _ .#. ];
# Final *k → *g after vowels (e.g. *drukos → drwg, *mrakis → brag).
define BritVowelFinalK [ {*k} -> {*g} || PCeltStarVowel _ .#. ];
define BritLenInterG [ {*g} -> {*ɣ} || PCeltStarVowel _ PCeltStarVowel ];
# Final *t → *d only after SHORT vowels (e.g. *bitus → bid, not *brātus → braud).
define BritLenFinalT [ {*t} -> {*d} || PCeltStarShortVowel _ .#. ];
# Final *t → *θ after *r or *l (e.g. *nertom → nerθ, *slattā → laθ).
define BritFinalTSpirRL [ {*t} -> {*θ} || [{*r} | {*l}] _ .#. ];
# Final *nd → *n: word-final *d after nasal is lost (e.g. *sindos → hin, *landā → lan).
define BritFinalNDrop [ {*n} {*d} -> {*n} || _ .#. ];
# *e → *i before *g: vowel raising before velars (e.g. *tegos → tig in Old Welsh).
define BritEtoIBeforeG [ {*e} -> {*i} || _ {*g} ];
# Final *ng > *ŋ: velar nasal coalescence at word end (*longā > loŋg/ɬoŋ).
define BritFinalNGCoal [ {*n} {*g} -> {*ŋ} || _ .#. ];
define BritLenition [
    BritSNSimpl
    .o. BritMedialXS .o. BritFinalNGCoal .o. BritFinalNDrop .o. BritEtoIBeforeG
    .o. BritLenInterD .o. BritLenFinalD .o. BritLenT .o. BritFinalTSpirRL .o. BritLenFinalT .o. BritLenFinalK .o. BritVowelFinalK .o. BritLenInterG
];


define BritRemoveStars [
    {*a}  -> a,      {*ā}  -> ({au} | a | o),
    {*e}  -> (e | ē),  {*ē}  -> ē,
    {*i}  -> i,      {*ī}  -> ī,
    {*o}  -> o,      {*ū}  -> ū,
    {*u}  -> (u | ū),
    {*oi} -> {oi},   {*aw} -> {aw},
    {*b}  -> b,      {*d}  -> (d | t),
    {*f}  -> f,      {*g}  -> (g | ɣ | 0),
    {*gw} -> g u,    {*h}  -> h,
    {*k}  -> k,      {*l}  -> (l | ɬ),
    {*m}  -> m,      {*n}  -> n,
    {*p}  -> (p | b),  {*r}  -> r,
    {*s}  -> s,      {*t}  -> t,
    {*v}  -> v,      {*w}  -> (u | w),
    {*x}  -> x,      {*y}  -> 0,
    {*θ}  -> θ,      {*ð}  -> (ð | d),
    {*ɣ}  -> (ɣ | 0),      {*χ}  -> (χ | 0),
    {*ŋ}  -> ŋ
];

define BritRules [
    PCeltInitialUPhi
    .o. BritSkwCluster
    .o. BritLabioVelar
    .o. BritGdCluster
    .o. BritInitialSClusters
    .o. PCeltPhiLoss
    .o. BritInitialS
    .o. BritInitialW
    .o. BritMedialWDrop
    .o. BritYDrop
    .o. BritMbSimpl
    .o. BritUtoI
    .o. BritVowelDiphthongs
    .o. BritIAffection
    .o. BritApocope
    .o. BritLenition
    .o. BritRemoveStars
];

define BritReflexes CelticProtoInput .o. BritRules;

## Gaulish transducer                        ##

# Gaulish is a Continental P-Celtic language that preserves endings more
# fully than Insular Celtic. Whether a particular form retains its final
# syllable depends on the declension and register; no general apocope rule
# is applied here — such forms will mismatch the TSV rather than
# over-deleting correct endings.

# 0g. Gaulish-specific treatment of initial *u before *ɸ: *u → *v, not *w.
#     (*uɸo > vo, *uɸer > ver). Applied before PCeltPhiLoss.
define GaulInitialUPhi [ {*u} -> {*v} || .#. _ {*ɸ} ];

# 1. *kʷ > *p, *gʷ > *b (P-Celtic).
define GaulLabioVelar [
    {*kʷ} -> {*p},
    {*gʷ} -> {*b}
];

# 2. Initial *sr- > *fr- (attested in *srognā > frogna; shared with Brittonic).
define GaulInitialSR [ {*s} {*r} -> {*f} {*r} || .#. _ ];

# 3. *w > *u everywhere (*wiros > uiros, *windos > uindus).
define GaulWVocal [ {*w} -> {*u} ];

# 4. *y > 0.
define GaulYDrop [ {*y} -> 0 ];

# 5. Diphthong reductions.
define GaulVowelDiphthongs [
    {*ei} -> {*e},
    {*ai} -> {*e},
    {*au} -> {*o},
    {*ou} -> {*u},
    {*ow} -> {*u}
];

# 6. Strip asterisks; macron notation for Gaulish long vowels.
define GaulRemoveStars [
    {*a}  -> a,   {*ā}  -> ā,
    {*e}  -> e,   {*ē}  -> ē,
    {*i}  -> i,   {*ī}  -> ī,
    {*o}  -> o,   {*ū}  -> ū,
    {*u}  -> u,
    {*b}  -> b,   {*d}  -> d,
    {*f}  -> f,   {*g}  -> g,
    {*k}  -> k,   {*l}  -> l,
    {*m}  -> m,   {*n}  -> n,
    {*p}  -> p,   {*r}  -> r,
    {*s}  -> s,   {*t}  -> t,
    {*v}  -> v,   {*w}  -> u,
    {*x}  -> x,   {*y}  -> 0,
    {*ŋ}  -> ŋ,   {*ɣ}  -> ɣ
];

define GaulRules [
    GaulInitialUPhi
    .o. PCeltPhiLoss
    .o. GaulLabioVelar
    .o. GaulInitialSR
    .o. GaulWVocal
    .o. GaulYDrop
    .o. GaulVowelDiphthongs
    .o. GaulRemoveStars
];

define GaulReflexes CelticProtoInput .o. GaulRules;
define Gaulish      GaulReflexes .o. GaulSurface;

# Celtiberian is a Q-Celtic Continental language. The TSV data (11 cognate
# sets) show conservative preservation with no length marking.

# 1. *kʷ > k+u, *gʷ > g+u (Q-Celtic labiovelars preserved as clusters).
define CeltibLabioVelar [
    {*kʷ} -> {*k} {*u},
    {*gʷ} -> {*g} {*u}
];

# 2. *w > *u everywhere.
define CeltibWVocal [ {*w} -> {*u} ];

# 3. Initial *d > *t (devoicing attested in *deiwos > teiuo).
define CeltibInitialD [ {*d} -> {*t} || .#. _ ];

# 4. *x > 0 (*rextus > retu).
define CeltibXLoss [ {*x} -> 0 ];

# 5. *y > 0.
define CeltibYDrop [ {*y} -> 0 ];

# 5a. Diphthong reductions (*au > *o as in *baus > bo).
define CeltibDiphthongs [ {*au} -> {*o}, {*ou} -> {*u} ];

# 5b. Final apocope: drop *s after vowel (*lugus > lugu, *deiwos > teiuo).
define CeltibApocope [ {*s} -> 0 || PCeltStarVowel _ .#. ];

# 6.
define CeltibRemoveStars [
    {*a}  -> a,   {*ā}  -> a,
    {*e}  -> e,   {*ē}  -> e,
    {*i}  -> i,   {*ī}  -> i,
    {*o}  -> o,
    {*u}  -> u,   {*ū}  -> u,
    {*b}  -> b,   {*d}  -> d,
    {*g}  -> g,   {*k}  -> k,
    {*l}  -> l,   {*m}  -> m,
    {*n}  -> n,   {*p}  -> p,
    {*r}  -> r,   {*s}  -> s,
    {*t}  -> t,   {*v}  -> v,
    {*w}  -> u,   {*x}  -> 0,
    {*y}  -> 0,   {*ŋ}  -> n,
    {*ei} -> e i, {*ai} -> a i,
    {*au} -> a u, {*ou} -> o u,
    {*ow} -> o u
];

define CeltibRules [
    PCeltPhiLoss
    .o. CeltibLabioVelar
    .o. CeltibWVocal
    .o. CeltibInitialD
    .o. CeltibXLoss
    .o. CeltibYDrop
    .o. CeltibDiphthongs
    .o. CeltibApocope
    .o. CeltibRemoveStars
];
